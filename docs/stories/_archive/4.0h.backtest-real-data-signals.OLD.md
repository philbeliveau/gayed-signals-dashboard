### Keep It Simple

**What we're building:**
```typescript
async function backtestSignal(
  signal: 'utilities_spy' | 'lumber_gold',
  startDate: string,
  endDate: string
) {
  // 1. Get real data
  const data = await unifiedDataService.fetchMarketData(
    ['SPY', 'XLU', 'WOOD', 'GLD'],
    { startDate, endDate }
  );

  // 2. Calculate signals
  const signals = await calculateSignals(signal, data);

  // 3. Simulate trades
  const trades = [];
  let position = null;

  for (const day of signals) {
    if (day.signal === 'Risk-On' && position !== 'SPY') {
      trades.push({ date: day.date, action: 'buy', symbol: 'SPY' });
      position = 'SPY';
    }
    if (day.signal === 'Risk-Off' && position !== 'XLU') {
      trades.push({ date: day.date, action: 'buy', symbol: 'XLU' });
      position = 'XLU';
    }
  }

  // 4. Calculate performance
  const totalReturn = calculateReturn(trades, data);
  const maxDrawdown = calculateDrawdown(trades, data);

  return { trades, totalReturn, maxDrawdown };
}
```
