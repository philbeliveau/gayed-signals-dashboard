## Status
- **Epic**: 4.0 Data Pipeline Restructuring

- **Priority**: High
- **Estimated Effort**: 10 points
- **Assigned To**: James (Dev Agent)

## Story

**As a** quantitative developer
**I want** all signal calculators deployed on Railway backend to follow a standardized interface and calculation pattern using unified data from Railway PostgreSQL
**So that** signals are computed consistently, are testable, maintainable, and can be easily extended with new signal types

## Background

Current signal calculation issues:
- Inconsistent calculation interfaces across different signals
- No standardized error handling
- Duplicated data fetching and validation logic
- No systematic caching or memoization
- Difficult to test calculations in isolation
- No standard metadata or provenance tracking
- Hard to add new signals without code duplication

This story creates a base calculator framework that all signals will extend, ensuring consistency, testability, and maintainability.

### BaseSignalCalculator Architecture

```typescript
/**
 * Abstract base class for all signal calculators
 * Deployed on Railway backend with PostgreSQL and Redis integration
 */
export abstract class BaseSignalCalculator<TInput, TOutput, TParams = any> {
  protected config: SignalConfig<TParams>;
  protected validator: DataQualityValidator<TInput>;
  protected prisma: PrismaClient;  // Railway PostgreSQL
  protected redis: Redis;          // Railway Redis
  protected logger: Logger;

  constructor(
    config: SignalConfig<TParams>,
    dependencies: CalculatorDependencies
  ) {
    this.config = config;
    this.validator = dependencies.validator;
    this.prisma = dependencies.prisma;
    this.redis = dependencies.redis;
    this.logger = Logger.getInstance();

    // Load configuration from Railway PostgreSQL
    this.validateConfig();
  }

  /**
   * Main entry point for signal calculation
   * Orchestrates the full lifecycle: fetch -> validate -> calculate -> persist
   */
  async compute(context: CalculationContext): Promise<SignalResult<TOutput>> {
    const startTime = Date.now();

    try {
      this.logger.info(`Starting ${this.config.name} calculation`, { context });

      // 1. Check cache first
      const cached = await this.checkCache(context);
      if (cached && !context.forceRecalculate) {
        this.logger.info('Returning cached result', { context });
        return cached;
      }

      // 2. Fetch required data
      const rawData = await this.fetchData(context);

      // 3. Validate input data
      const validationResult = await this.validateData(rawData, context);
      if (!validationResult.isValid()) {
        throw new ValidationError(
          `Data validation failed for ${this.config.name}`,
          validationResult
        );
      }

      // 4. Transform/prepare data
      const preparedData = await this.prepareData(rawData, context);

      // 5. Perform calculation
      const calculationResult = await this.calculate(preparedData, context);

      // 6. Post-process result (add metadata, format, etc.)
      const processedResult = await this.postProcess(
        calculationResult,
        preparedData,
        context
      );

      // 7. Persist result
      await this.persistResult(processedResult, context);

      // 8. Update cache
      await this.updateCache(processedResult, context);

      // 9. Record metrics
      const duration = Date.now() - startTime;
      await this.recordMetrics(duration, processedResult, context);

      this.logger.info(`Completed ${this.config.name} calculation`, {
        context,
        duration,
        result: processedResult
      });

      return processedResult;

    } catch (error) {
      this.logger.error(`Error in ${this.config.name} calculation`, {
        context,
        error
      });

      // Handle error and return degraded result if possible
      return await this.handleError(error, context);
    }
  }

  /**
   * Batch calculation for multiple contexts (dates, symbols, etc.)
   * Optimized for parallel execution
   */
  async computeBatch(contexts: CalculationContext[]): Promise<SignalResult<TOutput>[]> {
    // Pre-fetch all required data in bulk
    const dataMap = await this.fetchDataBatch(contexts);

    // Calculate in parallel with concurrency limit
    const concurrencyLimit = 5;
    const results: SignalResult<TOutput>[] = [];

    for (let i = 0; i < contexts.length; i += concurrencyLimit) {
      const batch = contexts.slice(i, i + concurrencyLimit);
      const batchResults = await Promise.all(
        batch.map(context => this.compute(context))
      );
      results.push(...batchResults);
    }

    return results;
  }

  /**
   * Incremental calculation - only calculate for new/updated data
   */
  async computeIncremental(context: CalculationContext): Promise<SignalResult<TOutput>> {
    const latestResult = await this.repository.getLatestSignal(this.config.name);

    if (!latestResult) {
      // No previous calculation, do full calculation
      return this.compute(context);
    }

    // Check if new data is available
    const hasNewData = await this.hasNewDataSince(latestResult.calculation_date);
    if (!hasNewData) {
      this.logger.info('No new data available, returning latest result');
      return this.mapToSignalResult(latestResult);
    }

    // Perform incremental calculation
    return this.compute(context);
  }

  // ===== Abstract Methods (must be implemented by subclasses) =====

  /**
   * Fetch raw data required for calculation
   */
  protected abstract fetchData(context: CalculationContext): Promise<TInput>;

  /**
   * Core calculation logic - the "heart" of the signal
   */
  protected abstract calculate(
    data: TInput,
    context: CalculationContext
  ): Promise<TOutput>;

  /**
   * Signal-specific metadata
   */
  protected abstract getSignalMetadata(): SignalMetadata;

  // ===== Default Implementations (can be overridden) =====

  /**
   * Validate configuration parameters
   */
  protected validateConfig(): void {
    if (!this.config.name) {
      throw new Error('Signal name is required');
    }
    if (!this.config.type) {
      throw new Error('Signal type is required');
    }
    // Subclasses can override to add parameter validation
  }

  /**
   * Validate input data before calculation
   */
  protected async validateData(
    data: TInput,
    context: CalculationContext
  ): Promise<ValidationResult> {
    return this.validator.validate(data, {
      signal: this.config.name,
      date: context.date,
      ...context.validationOptions
    });
  }

  /**
   * Prepare/transform data before calculation
   * Default: return data as-is
   */
  protected async prepareData(
    data: TInput,
    context: CalculationContext
  ): Promise<TInput> {
    return data;
  }

  /**
   * Post-process calculation result
   * Default: add standard metadata
   */
  protected async postProcess(
    result: TOutput,
    inputData: TInput,
    context: CalculationContext
  ): Promise<SignalResult<TOutput>> {
    const metadata = this.getSignalMetadata();

    return {
      signalName: this.config.name,
      signalType: this.config.type,
      calculationDate: context.date,
      calculationTimestamp: new Date(),
      value: result,
      metadata: {
        ...metadata,
        version: this.config.version || '1.0.0',
        parameters: this.config.parameters,
        dataQuality: await this.assessDataQuality(inputData),
        calculationDuration: Date.now() - context.startTime
      }
    };
  }

  /**
   * Persist calculation result
   */
  protected async persistResult(
    result: SignalResult<TOutput>,
    context: CalculationContext
  ): Promise<void> {
    await this.repository.insertSignal({
      signal_name: result.signalName,
      signal_type: result.signalType,
      calculation_date: result.calculationDate,
      signal_value: this.extractNumericValue(result.value),
      signal_strength: this.calculateStrength(result.value),
      signal_status: this.determineStatus(result.value),
      confidence_score: result.metadata.confidence,
      data_quality_score: result.metadata.dataQuality,
      input_data: JSON.stringify(context.inputSummary),
      calculation_params: JSON.stringify(this.config.parameters),
      calculation_version: this.config.version || '1.0.0'
    });
  }

  /**
   * Handle calculation errors
   */
  protected async handleError(
    error: Error,
    context: CalculationContext
  ): Promise<SignalResult<TOutput>> {
    // Log error with full context
    await this.repository.insertSignal({
      signal_name: this.config.name,
      signal_type: this.config.type,
      calculation_date: context.date,
      signal_value: null,
      signal_status: 'error',
      confidence_score: 0,
      data_quality_score: 0,
      calculation_params: JSON.stringify(this.config.parameters),
      error_message: error.message
    });

    throw error; // Re-throw after logging
  }

  /**
   * Check cache for existing result
   */
  protected async checkCache(
    context: CalculationContext
  ): Promise<SignalResult<TOutput> | null> {
    const cacheKey = this.getCacheKey(context);
    return this.cache.get(cacheKey);
  }

  /**
   * Update cache with new result
   */
  protected async updateCache(
    result: SignalResult<TOutput>,
    context: CalculationContext
  ): Promise<void> {
    const cacheKey = this.getCacheKey(context);
    const ttl = this.config.cacheTtl || 3600; // 1 hour default
    await this.cache.set(cacheKey, result, ttl);
  }

  /**
   * Generate cache key
   */
  protected getCacheKey(context: CalculationContext): string {
    return `signal:${this.config.name}:${context.date.toISOString()}:${JSON.stringify(this.config.parameters)}`;
  }

  /**
   * Fetch data for batch calculation
   */
  protected async fetchDataBatch(
    contexts: CalculationContext[]
  ): Promise<Map<string, TInput>> {
    // Default implementation: fetch individually
    // Subclasses can override for optimized bulk fetching
    const dataMap = new Map<string, TInput>();

    for (const context of contexts) {
      const data = await this.fetchData(context);
      dataMap.set(context.date.toISOString(), data);
    }

    return dataMap;
  }

  /**
   * Check if new data is available since a date
   */
  protected async hasNewDataSince(date: Date): Promise<boolean> {
    // Default implementation
    // Subclasses can override with data-source-specific logic
    return true;
  }

  /**
   * Assess overall data quality
   */
  protected async assessDataQuality(data: TInput): Promise<number> {
    // Return 0-1 score based on validation results
    const validationResult = await this.validator.validate(data, {
      signal: this.config.name
    });

    const totalViolations =
      validationResult.getErrors().length * 3 +
      validationResult.getWarnings().length;

    // Simple scoring: penalize violations
    return Math.max(0, 1 - (totalViolations * 0.1));
  }

  /**
   * Extract numeric value from result (for storage)
   */
  protected abstract extractNumericValue(result: TOutput): number;

  /**
   * Calculate signal strength (-1 to 1 or 0 to 1)
   */
  protected abstract calculateStrength(result: TOutput): number;

  /**
   * Determine signal status (bullish, bearish, neutral, etc.)
   */
  protected abstract determineStatus(result: TOutput): SignalStatus;

  /**
   * Record performance metrics
   */
  protected async recordMetrics(
    duration: number,
    result: SignalResult<TOutput>,
    context: CalculationContext
  ): Promise<void> {
    // Record to metrics system (Prometheus, StatsD, etc.)
    // Implementation depends on monitoring infrastructure
  }
}
```

```typescript
export interface Gayed8MonthParams {
  shortPeriod: number; // Default: 8
  longPeriod: number;  // Default: 9
  minimumDataPoints: number; // Default: 20
}

export interface Gayed8MonthInput {
  spy: OHLCData[];
  date: Date;
}

export interface Gayed8MonthOutput {
  shortMA: number;
  longMA: number;
  spread: number;
  signal: 'bullish' | 'bearish';
  confidence: number;
}

export class Gayed8MonthCalculator extends BaseSignalCalculator<
  Gayed8MonthInput,
  Gayed8MonthOutput,
  Gayed8MonthParams
> {

  protected async fetchData(context: CalculationContext): Promise<Gayed8MonthInput> {
    const endDate = context.date;
    const startDate = new Date(endDate);
    startDate.setMonth(startDate.getMonth() - 12); // Get 12 months of data

    const spy = await this.marketDataRepository.getMarketData(
      'SPY',
      startDate,
      endDate,
      'price'
    );

    return {
      spy,
      date: context.date
    };
  }

  protected async calculate(
    data: Gayed8MonthInput,
    context: CalculationContext
  ): Promise<Gayed8MonthOutput> {
    const { spy, date } = data;
    const { shortPeriod, longPeriod } = this.config.parameters;

    // Get data up to calculation date
    const relevantData = spy.filter(d => d.date <= date);

    if (relevantData.length < this.config.parameters.minimumDataPoints) {
      throw new Error(`Insufficient data: ${relevantData.length} points, need ${this.config.parameters.minimumDataPoints}`);
    }

    // Calculate moving averages
    const shortMA = this.calculateMA(relevantData, shortPeriod);
    const longMA = this.calculateMA(relevantData, longPeriod);

    // Calculate spread
    const spread = ((shortMA - longMA) / longMA) * 100;

    // Determine signal
    const signal = shortMA > longMA ? 'bullish' : 'bearish';

    // Calculate confidence based on spread magnitude
    const confidence = Math.min(1, Math.abs(spread) / 5); // 5% spread = 100% confidence

    return {
      shortMA,
      longMA,
      spread,
      signal,
      confidence
    };
  }

  protected getSignalMetadata(): SignalMetadata {
    return {
      description: 'Gayed 8-Month Timing Signal',
      interpretation: 'Bullish when 8-month MA > 9-month MA, Bearish otherwise',
      thresholds: {
        strongBullish: 2.0,  // > 2% spread
        weakBullish: 0.5,    // > 0.5% spread
        neutral: 0,
        weakBearish: -0.5,
        strongBearish: -2.0
      },
      recommendedUse: 'Monthly rebalancing, trend following strategy',
      historicalAccuracy: 0.73  // 73% historical accuracy
    };
  }

  protected extractNumericValue(result: Gayed8MonthOutput): number {
    return result.spread;
  }

  protected calculateStrength(result: Gayed8MonthOutput): number {
    // Normalize spread to -1 to 1 range
    // Clip at +/- 10% spread
    return Math.max(-1, Math.min(1, result.spread / 10));
  }

  protected determineStatus(result: Gayed8MonthOutput): SignalStatus {
    if (result.spread > 2.0) return 'bullish';
    if (result.spread > 0.5) return 'bullish';
    if (result.spread < -2.0) return 'bearish';
    if (result.spread < -0.5) return 'bearish';
    return 'neutral';
  }

  private calculateMA(data: OHLCData[], period: number): number {
    const recent = data.slice(-period);
    const sum = recent.reduce((acc, d) => acc + d.close, 0);
    return sum / period;
  }
}
```

### Bollinger Band Signal Calculator

```typescript
export interface BollingerBandParams {
  period: number;           // Default: 20
  standardDeviations: number; // Default: 2
  overboughtThreshold: number; // Default: 0.8
  oversoldThreshold: number;   // Default: 0.2
}

export interface BollingerBandInput {
  prices: number[];
  date: Date;
}

export interface BollingerBandOutput {
  upperBand: number;
  middleBand: number;
  lowerBand: number;
  currentPrice: number;
  percentB: number; // Position within bands (0 to 1)
  bandwidth: number;
  signal: 'overbought' | 'oversold' | 'neutral';
  confidence: number;
}

export class BollingerBandCalculator extends BaseSignalCalculator<
  BollingerBandInput,
  BollingerBandOutput,
  BollingerBandParams
> {

  protected async fetchData(context: CalculationContext): Promise<BollingerBandInput> {
    const symbol = context.symbol || 'SPY';
    const endDate = context.date;
    const startDate = new Date(endDate);
    startDate.setDate(startDate.getDate() - (this.config.parameters.period * 2));

    const marketData = await this.marketDataRepository.getMarketData(
      symbol,
      startDate,
      endDate,
      'price'
    );

    return {
      prices: marketData.map(d => d.close),
      date: context.date
    };
  }

  protected async calculate(
    data: BollingerBandInput,
    context: CalculationContext
  ): Promise<BollingerBandOutput> {
    const { period, standardDeviations, overboughtThreshold, oversoldThreshold } =
      this.config.parameters;

    const prices = data.prices;
    const currentPrice = prices[prices.length - 1];

    // Calculate middle band (SMA)
    const recentPrices = prices.slice(-period);
    const middleBand = recentPrices.reduce((a, b) => a + b, 0) / period;

    // Calculate standard deviation
    const squaredDiffs = recentPrices.map(p => Math.pow(p - middleBand, 2));
    const variance = squaredDiffs.reduce((a, b) => a + b, 0) / period;
    const stdDev = Math.sqrt(variance);

    // Calculate bands
    const upperBand = middleBand + (standardDeviations * stdDev);
    const lowerBand = middleBand - (standardDeviations * stdDev);

    // Calculate %B (position within bands)
    const percentB = (currentPrice - lowerBand) / (upperBand - lowerBand);

    // Calculate bandwidth (volatility indicator)
    const bandwidth = ((upperBand - lowerBand) / middleBand) * 100;

    // Determine signal
    let signal: 'overbought' | 'oversold' | 'neutral';
    if (percentB > overboughtThreshold) {
      signal = 'overbought';
    } else if (percentB < oversoldThreshold) {
      signal = 'oversold';
    } else {
      signal = 'neutral';
    }

    // Calculate confidence
    const distanceFromThreshold = signal === 'overbought'
      ? percentB - overboughtThreshold
      : signal === 'oversold'
      ? oversoldThreshold - percentB
      : 0;

    const confidence = signal === 'neutral'
      ? 0.5
      : Math.min(1, distanceFromThreshold * 5);

    return {
      upperBand,
      middleBand,
      lowerBand,
      currentPrice,
      percentB,
      bandwidth,
      signal,
      confidence
    };
  }

  protected getSignalMetadata(): SignalMetadata {
    return {
      description: 'Bollinger Band Volatility Signal',
      interpretation: 'Overbought when price near upper band, oversold when near lower band',
      thresholds: {
        overbought: this.config.parameters.overboughtThreshold,
        oversold: this.config.parameters.oversoldThreshold
      },
      recommendedUse: 'Mean reversion strategy, volatility analysis'
    };
  }

  protected extractNumericValue(result: BollingerBandOutput): number {
    return result.percentB;
  }

  protected calculateStrength(result: BollingerBandOutput): number {
    // Convert percentB (0-1) to strength (-1 to 1)
    // 0 = oversold (-1), 0.5 = neutral (0), 1 = overbought (1)
    return (result.percentB * 2) - 1;
  }

  protected determineStatus(result: BollingerBandOutput): SignalStatus {
    if (result.signal === 'overbought') return 'defensive';
    if (result.signal === 'oversold') return 'bullish';
    return 'neutral';
  }
}
```
