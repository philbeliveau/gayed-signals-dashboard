### Task 2: Railway API Client Implementation (AC: 2)
**Estimated Effort:** 45 minutes

- [x] **Subtask 2.1:** Create Railway HTTP client
  ```typescript
  // apps/web/src/lib/api/railway-client.ts
  import { RailwayConfig, RailwayResponse } from './types';

  export class RailwayClient {
    private baseURL: string;
    private apiKey: string;
    private timeout: number = 10000;

    constructor(config?: Partial<RailwayConfig>) {
      this.baseURL = config?.baseURL || process.env.NEXT_PUBLIC_RAILWAY_BACKEND_URL!;
      this.apiKey = config?.apiKey || process.env.RAILWAY_API_KEY!;
    }

    async request<T>(endpoint: string, options?: RequestInit): Promise<RailwayResponse<T>> {
      const url = `${this.baseURL}${endpoint}`;

      try {
        const response = await fetch(url, {
          ...options,
          headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${this.apiKey}`,
            ...options?.headers,
          },
          signal: AbortSignal.timeout(this.timeout),
        });

        if (!response.ok) {
          throw new Error(`Railway API error: ${response.status}`);
        }

        return await response.json();
      } catch (error) {
        console.error('Railway client error:', error);
        throw error;
      }
    }
  }
  ```

### Task 3: V2 API Signal Service (AC: 3)

- [x] **Subtask 3.1:** Create typed wrapper for `/api/v2/signals`
  ```typescript
  // apps/web/src/lib/api/signals-v2.ts
  import { RailwayClient } from './railway-client';
  import { SignalV2Response, SignalQueryParams } from './types';

  export class SignalsV2Service {
    private client: RailwayClient;

    constructor() {
      this.client = new RailwayClient();
    }

    async getSignals(params?: SignalQueryParams): Promise<SignalV2Response> {
      const queryString = this.buildQueryString(params);
      return this.client.request<SignalV2Response>(`/api/v2/signals${queryString}`);
    }

    private buildQueryString(params?: SignalQueryParams): string {
      if (!params) return '';
      const searchParams = new URLSearchParams();
      if (params.fast) searchParams.set('fast', 'true');
      if (params.symbols) searchParams.set('symbols', params.symbols.join(','));
      return searchParams.toString() ? `?${searchParams.toString()}` : '';
    }
  }
  ```

### Task 4: Adapter Layer for Backward Compatibility (AC: 4)
**Estimated Effort:** 1 hour

- [x] **Subtask 4.1:** Create V2 → Legacy adapter
  ```typescript
  // apps/web/src/lib/adapters/signals-adapter.ts
  import { SignalV2Response, SignalV2 } from '@/lib/api/types';
  import { Signal, ConsensusSignal } from '@/types/signals';

  export class SignalsAdapter {
    /**
     * Transform V2 API response to legacy format for UI compatibility
     */
    static toLegacyFormat(v2Response: SignalV2Response): { signals: Signal[], consensus: ConsensusSignal } {
      const signals = v2Response.data.map(this.transformSignal);
      const consensus = this.calculateConsensus(signals);

      return { signals, consensus };
    }

    private static transformSignal(v2Signal: SignalV2): Signal {
      return {
        type: v2Signal.type,
        signal: v2Signal.signal,
        strength: v2Signal.strength,
        confidence: v2Signal.confidence,
        rawValue: v2Signal.rawValue,
        date: v2Signal.date,
        provenance: v2Signal.provenance,
      };
    }

    private static calculateConsensus(signals: Signal[]): ConsensusSignal {
      const riskOnCount = signals.filter(s => s.signal === 'Risk-On').length;
      const riskOffCount = signals.filter(s => s.signal === 'Risk-Off').length;

      let consensus: 'Risk-On' | 'Risk-Off' | 'Mixed' = 'Mixed';
      if (riskOnCount > riskOffCount) consensus = 'Risk-On';
      if (riskOffCount > riskOnCount) consensus = 'Risk-Off';

      const avgConfidence = signals.reduce((sum, s) => sum + s.confidence, 0) / signals.length;

      return {
        consensus,
        confidence: avgConfidence,
        riskOnCount,
        riskOffCount,
        signals,
      };
    }
  }
  ```

### Data Pipeline Patterns (from coding-standards.md)

**ALWAYS use Railway backend for data fetching:**
```typescript
// ✅ CORRECT: Use Railway V2 API
const v2Service = new SignalsV2Service();
const response = await v2Service.getSignals();

// ❌ WRONG: Direct fetch to old API
const response = await fetch('/api/signals');
```

**Graceful degradation required:**
```typescript
// ✅ CORRECT: Try Railway, fallback to local
try {
  return await railwayClient.getSignals();
} catch (error) {
  console.warn('Railway unavailable, using local API');
  return await fetch('/api/signals').then(r => r.json());
}

// ❌ WRONG: No fallback handling
return await railwayClient.getSignals();
```

**Feature flag pattern:**
